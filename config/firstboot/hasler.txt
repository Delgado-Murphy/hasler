# Hasler: settings for a new radio. Copy this file to the SD card's boot

# Which releases the radio follows: stable (everyone) or bench (the makers'
# test radio - it gets every release first).
channel = stable

# Updates: auto (small ones at night, big ones ask first), ask, or off.
update_mode = auto

# The cabinet: hasler-hm-hat (a Hasler HM on the Hasler HAT - every new one),
# hasler-hm (the prototype's stripboard) or excelsior.
variant = hasler-hm-hat

# The built-in stations: the starter set on the keys out of the box. One of
# eclectic, classical, jazz, pop, raices, family (config/presets in the
# release); the owner can pick another on the key page.
starter = eclectic

# The login you set in Raspberry Pi Imager. Empty = the first user (uid 1000).
user = pi

# Give moOde the Hasler audio and casting settings (Amp4 Pro, AirPlay,
# Bluetooth, UPnP, names "Hasler"). no = leave moOde as it is.
moode = yes

# yes = may run without the cabinet's key board (a bare Pi on the bench).
